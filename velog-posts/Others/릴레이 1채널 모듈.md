<h2 id="기본-예제">기본 예제</h2>
<pre><code class="language-cpp">int relay = 5;

void setup() {
  pinMode(relay, OUTPUT);
  digitalWrite(relay, LOW);
}

void loop() {
  digitalWrite(relay, HIGH);
  delay(1000);
  digitalWrite(relay, LOW);
  delay(1000);
}</code></pre>
