<h1 id="습도-센서dht11-실습_라이브러리-사용">습도 센서(DHT11) 실습_라이브러리 사용</h1>
<pre><code class="language-c">#include &quot;DHT.h&quot;          // DHT센서 라이브러리가 포함된 헤더파일 추가 필요

#define DHTPIN 2          // 2번 핀에 연결
#define DHTTYPE DHT11

// Initialize DHT sensor.
DHT dht(DHTPIN, DHTTYPE);

void setup() {
  Serial.begin(9600);     // 시리얼 통신 9600bps로 초기화
  dht.begin();
}

void loop() {
  delay(2000);            // 센서 측정 주기(약 1~2초)에 맞춰
                          // 딜레이 삽입(안정적 측정을 위해)

  float h = dht.readHumidity();        // 습도
  float t = dht.readTemperature();     // 온도(°C)
  float f = dht.readTemperature(true); // 온도(°F)

  // Check if any reads failed and exit early (to try again).
  if (isnan(h) || isnan(t) || isnan(f)) {
    Serial.println(F(&quot;Failed to read from DHT sensor!&quot;));
    return;
  }

  // 열 지수(체감온도) 계산
    float hif = dht.computeHeatIndex(f, h);         // 화씨
  float hic = dht.computeHeatIndex(t, h, false);  // 섭씨

  Serial.print(F(&quot;Humidity: &quot;));
  Serial.print(h);
  Serial.print(F(&quot;%  Temperature: &quot;));
  Serial.print(t);
  Serial.print(F(&quot;°C &quot;));
  Serial.print(f);
  Serial.print(F(&quot;°F  Heat index: &quot;));
  Serial.print(hic);
  Serial.print(F(&quot;°C &quot;));
  Serial.print(hif);
  Serial.println(F(&quot;°F&quot;));
}</code></pre>
