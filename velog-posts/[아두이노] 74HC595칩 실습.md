<h3 id="74hc595">74HC595</h3>
<ul>
<li><strong>8비트 시프트 레지스터</strong>: 직렬로 데이터를 입력 받아 8비트 병렬 출력으로 변환.</li>
<li><strong>저전력 CMOS 기술</strong>: 2V~6V의 넓은 전원 전압 범위에서 동작.</li>
<li><strong>3상태 출력(3-State Outputs)</strong>: 출력 핀을 High, Low, 또는 고임피던스(High Impedance) 상태로 설정 가능.</li>
<li><strong>캐스케이딩 가능</strong>: 여러 74HC595를 연결해 출력 핀을 무제한으로 확장 가능.</li>
<li><strong>최대 클럭 주파수</strong>: 4.5V에서 약 25MHz.</li>
<li><strong>낮은 전력 소비</strong>: 효율적인 전력 사용으로 배터리 구동 프로젝트에 적합.</li>
<li><strong>비동기 리셋</strong>: 쉬프트 레지스터를 초기화 가능.</li>
</ul>

<h3 id="기본-예제">기본 예제</h3>
<pre><code class="language-c">#include &lt;Arduino.h&gt;

int DataPin = 12; // 정수형 변수 선언 후 디지털 핀 번호 12번으로 초기화
int ClockPin = 11;
int LatchPin = 10;

// 정수형 2차원 배열 선언 후 숫자를 나타내는 8비트 배열로 초기화 
int DataArray[10] = {
  B11111100, // 0
  B01100000, // 1
  B11011010, // 2
  B11110010, // 3
  B01100110, // 4
  B10110110, // 5
  B10111110, // 6
  B11100000, // 7
  B11111110, // 8
  B11110110 // 9
};

void setup()
{
  pinMode(DataPin, OUTPUT); // 디지털 12번 핀을 출력(OUTPUT) 모드로 설정
  pinMode(ClockPin, OUTPUT);
  pinMode(LatchPin, OUTPUT);
}

void loop()
{
  // 7-세그먼트 디스플레이에 숫자를 순서대로 출력 
  for (int i = 0; i &lt; 10; i++)
  {
    digitalWrite(LatchPin, LOW); // 디지털 10번 핀에 디지털 신호 LOW 출력 
    // ClockPin에서의 상승 에지에 맞게 DataPin으로부터 입력 받은 DataArray[i]를 오른쪽 끝부터 전달
    shiftOut(DataPin, ClockPin, LSBFIRST, DataArray[i]); 
    digitalWrite(LatchPin, HIGH); // 디지털 10번 핀에 디지털 신호 HIGH 출력
    delay(1000);
  }
}</code></pre>

#include &quot;wiring_private.h&quot;

uint8_t shiftIn(uint8_t dataPin, uint8_t clockPin, uint8_t bitOrder) {
    uint8_t value = 0;
    uint8_t i;

    for (i = 0; i &lt; 8; ++i) {
        digitalWrite(clockPin, HIGH);
        if (bitOrder == LSBFIRST)
            value |= digitalRead(dataPin) &lt;&lt; i;
        else
            value |= digitalRead(dataPin) &lt;&lt; (7 - i);
        digitalWrite(clockPin, LOW);
    }
    return value;
}

void shiftOut(uint8_t dataPin, uint8_t clockPin, uint8_t bitOrder, uint8_t val)
{
    uint8_t i;

    for (i = 0; i &lt; 8; i++)  {
        if (bitOrder == LSBFIRST) {
            digitalWrite(dataPin, val &amp; 1);
            val &gt;&gt;= 1;
        } else {    
            digitalWrite(dataPin, (val &amp; 128) != 0);
            val &lt;&lt;= 1;
        }

        digitalWrite(clockPin, HIGH);
        digitalWrite(clockPin, LOW);    

        digitalWrite(10, HIGH); // 10 = LatchPin
        digitalWrite(10, LOW);    

        delay(300);    // shift 확인 위해 delay

    }
}</code></pre>
