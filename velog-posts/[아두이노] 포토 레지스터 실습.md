  Serial.begin(9600);

void loop() {
  int sensorValue = (analogRead(cdsPin));  // 0~1023 값 읽기
  float voltage = map((float)sensorValue,0,1023,0,5.0);
  Serial.print(&quot;Light value:&quot;);           // 값이 높을수록 밝음 낮을수록 어두움
