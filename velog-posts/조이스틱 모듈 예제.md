<pre><code class="language-c">#include &lt;Arduino.h&gt;

const int REDP = 11;  // select the pin for the red LED
const int GREENP = 5; // select the pin for the green LED
const int BLUEP = 6;  // select the pin for the blue LED

const int JOYZ = 2;

int joyz_tmp = 0;
int joyx_tmp = 1024 / 2 - 1; // A0
int joyy_tmp = 1024 / 2 - 1; // A1

int r_tmp = 254 / 2;
int g_tmp = 254 / 2;
int b_tmp = 1;

void setup() {
  // put your setup code here, to run once:
  pinMode(REDP, OUTPUT);

  pinMode(BLUEP, OUTPUT);

  pinMode(GREENP, OUTPUT);

  pinMode(JOYZ, INPUT_PULLUP);

  Serial.begin(9600);
}

void loop() {
  joyx_tmp = analogRead(A0);
  joyy_tmp = analogRead(A1);
  joyz_tmp = digitalRead(JOYZ);

  if(joyy_tmp &lt; 512){
    r_tmp = 0;
    g_tmp = (511-joyy_tmp)/2;
  }else{
    g_tmp = 0;
    r_tmp = (joyy_tmp-512)/2;
  }

  if(joyx_tmp &lt; 512){
    b_tmp = 0;
  }else{
    b_tmp = (joyx_tmp-512)/2;
  }

  analogWrite(REDP, r_tmp);
  analogWrite(BLUEP, b_tmp);
  analogWrite(GREENP, g_tmp);

  delay(1);
}</code></pre>
