<li>왼쪽  → GND(그라운드) 연결</li>

</ul>

<h2 id="피에조-부저--가변저항-응용-예제">피에조 부저 + 가변저항 응용 예제</h2>
<p><img alt="" src="https://velog.velcdn.com/images/mommers/post/37dac19b-7d39-49be-9c79-94659ed320ca/image.png" /></p>
<pre><code class="language-c">#include &lt;Arduino.h&gt;

int buzzerPin = 9;
int melody[] = {262, 294, 330, 349, 392, 440, 494, 523}; // 도레미파솔라시도
int num = 8; // 음계 개수

int lastNoteIndex = -1; // 처음 재생되는 음 X, 0번인 도의 출력 문제 방지

void setup() {
  Serial.begin(9600);
  pinMode(buzzerPin, OUTPUT);
}

void loop() {
  int val = analogRead(A0);
  int noteIndex = map(val, 0, 1023, 0, num-1); // 8개 이므로 0~7까지 사용

  if (noteIndex &lt; 0) noteIndex = 0; // 도가 출력이 안되는 것을 방지
  if (noteIndex &gt;= num){
        noteIndex = num - 1;
        }
        // 인덱스 범위를 벗어나지 않게 함 
        // 음이 나온 후 삐.. 소리 나는 것을 방지

  if (noteIndex != lastNoteIndex) { // -1이 아닐 경우 음 출력
    lastNoteIndex = noteIndex;      // 현재 음을 저장해, 비교하여 바뀌지 않으면 출력 X
    int freq = melody[noteIndex];

    tone(buzzerPin, freq);     // 소리 재생
    delay(1000);                // 1초 동안 소리 유지
    noTone(buzzerPin);         // 소리 끔
    delay(500);                // 다음 음으로 넘어가기 전 잠깐 대기

    // 시리얼 출력
    Serial.print(&quot;Note: &quot;);
    Serial.print(noteIndex);
    Serial.print(&quot;  Freq: &quot;);
    Serial.println(freq);
  }
}</code></pre>
